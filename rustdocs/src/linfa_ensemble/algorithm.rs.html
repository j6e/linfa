21/// A fitted ensemble of learners for classification.
22///
23/// ## Structure
24///
25/// An Ensemble Learner is composed of a collection of fitted models of type `M`.
26///
27/// ## Fitting Algorithm
28///
29/// Given a [DatasetBase](DatasetBase) denoted as `D`,
30/// 1. Create as many distinct bootstrapped subset of the original dataset `D` as number of
31///    distinct model to fit.
32/// 2. Fit each distinct model on a distinct bootstrapped subset of `D`.
33///
34/// Note that the subset size, as well as the subset of feature to use in each training subset can
35/// be specified in the [parameters](crate::EnsembleLearnerParams).
36///
37/// ## Prediction Algorithm
38///
39/// The prediction result is the result of majority voting across the fitted learners.
40///
41/// ## Example
42///
43/// This example shows how to train a bagging model using 100 decision trees,
44/// each trained on 70% of the training data (bootstrap sampling).
45/// ```no_run
46/// use linfa::prelude::{Fit, Predict};
47/// use linfa_ensemble::EnsembleLearnerParams;
48/// use linfa_trees::DecisionTree;
49/// use ndarray_rand::rand::SeedableRng;
50/// use rand::rngs::SmallRng;
51///
52/// // Load Iris dataset
53/// let mut rng = SmallRng::seed_from_u64(42);
54/// let (train, test) = linfa_datasets::iris()
55///     .shuffle(&mut rng)
56///     .split_with_ratio(0.8);
57///
58/// // Train the model on the iris dataset
59/// let bagging_model = EnsembleLearnerParams::new(DecisionTree::params())
60///     .ensemble_size(100)        // Number of Decision Tree to fit
61///     .bootstrap_proportion(0.7) // Select only 70% of the data via bootstrap
62///     .fit(&train)
63///     .unwrap();
64///
65/// // Make predictions on the test set
66/// let predictions = bagging_model.predict(&test);
67/// ```
68///
69/// ## References
70///
71/// * [Scikit-Learn User Guide](https://scikit-learn.org/stable/modules/ensemble.html)
72/// * [An Introduction to Statistical Learning](https://www.statlearning.com/)
73#[cfg_attr(

78pub struct EnsembleLearner<M> {
79    pub models: Vec<M>,
80    pub model_features: Vec<Vec<usize>>,
81}

83impl<M> EnsembleLearner<M> {
84    // Generates prediction iterator returning predictions from each model
85    pub fn generate_predictions<'b, R: Records, T>(
86        &'b self,
87        x: &'b [R],
88    ) -> impl Iterator<Item = T> + 'b
89    where
90        M: Predict<&'b R, T>,
91    {
92        self.models
93            .iter()
94            .zip(x.iter())
95            .map(move |(m, sub_data)| m.predict(sub_data))
96    }
97}

99impl<F: Clone, T, M> PredictInplace<Array2<F>, T> for EnsembleLearner<M>
100where
101    M: PredictInplace<Array2<F>, T>,
102    <T as AsTargets>::Elem: Copy + Eq + Hash + std::fmt::Debug,
103    T: AsTargets + AsTargetsMut<Elem = <T as AsTargets>::Elem>,
104{
105    fn predict_inplace(&self, x: &Array2<F>, y: &mut T) {
106        let y_array = y.as_targets();
107        assert_eq!(
108            x.nrows(),
109            y_array.len_of(Axis(0)),
110            "The number of data points must match the number of outputs."
111        );
112
113        let sub_datas = self
114            .model_features
115            .iter()
116            .map(|feat| x.select(Axis(1), feat))
117            .collect::<Vec<_>>();
118        let predictions = self.generate_predictions(&sub_datas);
119
120        // prediction map has same shape as y_array, but the elements are maps
121        let mut prediction_maps = y_array.map(|_| HashMap::new());
122
123        for prediction in predictions {
124            let p_arr = prediction.as_targets();
125            assert_eq!(p_arr.shape(), y_array.shape());
126            // Insert each prediction value into the corresponding map
127            Zip::from(&mut prediction_maps)
128                .and(&p_arr)
129                .for_each(|map, val| *map.entry(*val).or_insert(0) += 1);
130        }
131
132        // For each prediction, pick the result with the highest number of votes
133        let agg_preds = prediction_maps.map(|map| map.iter().max_by_key(|(_, v)| **v).unwrap().0);
134        let mut y_array = y.as_targets_mut();
135        for (y, pred) in y_array.iter_mut().zip(agg_preds.iter()) {
136            *y = **pred
137        }
138    }
139
140    fn default_target(&self, x: &Array2<F>) -> T {
141        self.models[0].default_target(x)
142    }
143}

145impl<D, T, P: Fit<Array2<D>, T::Owned, Error>, R: Rng + Clone> Fit<Array2<D>, T, Error>
146    for EnsembleLearnerValidParams<P, R>
147where
148    D: Clone,
149    T: FromTargetArrayOwned,
150    T::Elem: Copy + Eq + Hash,
151    T::Owned: AsTargets,
152{
153    type Object = EnsembleLearner<P::Object>;
154
155    fn fit(
156        &self,
157        dataset: &DatasetBase<Array2<D>, T>,
158    ) -> core::result::Result<Self::Object, Error> {
159        let mut models = Vec::with_capacity(self.ensemble_size);
160        let mut model_features = Vec::with_capacity(self.ensemble_size);
161        let mut rng = self.rng.clone();
162
163        // Compute dataset and the subset of features ratio to be selected
164        let dataset_size =
165            ((dataset.records.nrows() as f64) * self.bootstrap_proportion).ceil() as usize;
166        let n_feat = dataset.records.ncols();
167        let n_sub = ((n_feat as f64) * self.feature_proportion).ceil() as usize;
168
169        let iter = dataset.bootstrap_with_indices((dataset_size, n_sub), &mut rng);
170        for (train, _, feature_selected) in iter {
171            let model = self.model_params.fit(&train).unwrap();
172            models.push(model);
173            model_features.push(feature_selected);
174
175            if models.len() == self.ensemble_size {
176                break;
177            }
178        }
179
180        Ok(EnsembleLearner {
181            models,
182            model_features,
183        })
184    }
185}
